create index if not exists facturas_ia_hash_idx on public.facturas_ia (archivo_hash); -- 2) Búsqueda de duplicados para TODOS los usuarios (aunque su perfil no vea las facturas de otros). -- Devuelve solo lo necesario para el aviso: cuándo, quién, estado. create or replace function public.fac_buscar_duplicado(p_ruc text, p_nro text, p_hash text, p_excluir uuid default null) returns table(id uuid, estado text, creado_por text, creado_en timestamptz, emisor_razon text, emisor_ruc text, nro_factura text, motivo text) language sql stable security definer set search_path = public as $$ select f.id, f.estado, f.creado_por, f.creado_en, f.emisor_razon, f.emisor_ruc, f.nro_factura, case when coalesce(p_hash,'') <> '' and f.archivo_hash = p_hash then 'archivo' else 'numero' end from facturas_ia f where fac_ia_puede() and f.estado <> 'anulada' and (p_excluir is null or f.id <> p_excluir) and ( (coalesce(p_hash,'') <> '' and f.archivo_hash = p_hash) or (coalesce(p_nro,'') <> '' and f.nro_factura = p_nro and (coalesce(p_ruc,'') = '' or coalesce(f.emisor_ruc,'') = '' or f.emisor_ruc = p_ruc)) ) order by f.creado_en limit 5; $$; grant execute on function public.fac_buscar_duplicado(text,text,text,uuid) to authenticated; -- 3) Último timbrado conocido de un proveedor (para completar o verificar el que lee la IA) create or replace function public.fac_proveedor_ultimo(p_ruc text) returns table(emisor_razon text, timbrado text, timbrado_inicio date, timbrado_fin date) language sql stable security definer set search_path = public as $$ select f.emisor_razon, f.timbrado, f.timbrado_inicio, f.timbrado_fin from facturas_ia f where fac_ia_puede() and f.emisor_ruc = p_ruc and f.estado <> 'anulada' and coalesce(f.timbrado,'') <> '' order by f.fecha_emision desc nulls last, f.creado_en desc limit 1; $$; grant execute on function public.fac_proveedor_ultimo(text) to authenticated; -- 4) Regla en la base: no puede haber dos facturas activas con el mismo proveedor y Nº de comprobante. -- Si ya hubiera duplicados cargados, NO se crea el índice y se listan abajo para resolverlos (anular uno). do $$ begin if exists ( select 1 from facturas_ia where estado <> 'anulada' and coalesce(nro_factura,'') <> '' group by coalesce(emisor_ruc,''), nro_factura having count(*) > 1 ) then raise notice 'Hay facturas duplicadas: anulá las sobrantes y volvé a correr este script para activar la regla.'; else execute 'create unique index if not exists facturas_ia_unica on public.facturas_ia (coalesce(emisor_ruc,''''), nro_factura) where estado <> ''anulada'' and coalesce(nro_factura,'''') <> '''''; end if; end $$; -- Verificación: duplicados existentes (debería devolver 0 filas) y si la regla quedó activa select coalesce(emisor_ruc,'') as ruc, nro_factura, count(*) as veces, string_agg(estado||' '||to_char(creado_en,'DD/MM HH24:MI'), ' | ') as detalle from facturas_ia where estado <> 'anulada' and coalesce(nro_factura,'') <> '' group by 1,2 having count(*) > 1 union all select 'REGLA ACTIVA', (select case when exists(select 1 from pg_indexes where indexname='facturas_ia_unica') then 'sí' else 'NO' end), null, null;

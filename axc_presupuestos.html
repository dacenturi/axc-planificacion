create or replace function pg_temp.pp86_usos() returns table(tabla text, columna text, filas bigint, evaluacion text) language plpgsql as $$ declare r record; n bigint; begin for r in select c.table_name::text t, c.column_name::text col from information_schema.columns c join information_schema.tables tb on tb.table_schema = c.table_schema and tb.table_name = c.table_name where c.table_schema = 'public' and tb.table_type = 'BASE TABLE' and c.data_type in ('text','character varying') and c.table_name not like 'respaldo_%' and (c.column_name like '%presupuesto%' or (c.table_name in ('presupuesto_cab','presupuesto_edicion') and c.column_name = 'codigo')) loop execute format('select count(*) from public.%I where %I ~ %L', r.t, r.col, '^PP86([- ]|$)') into n; if n > 0 then tabla := r.t; columna := r.col; filas := n; evaluacion := case when r.t like 'presupuesto%' then 'propio del presupuesto: se borra' else 'OTRO MÓDULO: avisar antes de seguir' end; return next; end if; end loop; end $$; select * from pg_temp.pp86_usos() order by evaluacion desc, tabla;

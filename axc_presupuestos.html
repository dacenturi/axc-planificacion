-- 39_rubros_opcionales.sql -- AXC Constructora · Conexdata -- Rubros que el cliente puede elegir o descartar. No suman al total del -- presupuesto: van en un bloque aparte, con su propio subtotal. ALTER TABLE presupuesto_rubros ADD COLUMN IF NOT EXISTS opcional boolean NOT NULL DEFAULT false; COMMENT ON COLUMN presupuesto_rubros.opcional IS 'Rubro opcional para el cliente. Queda fuera del total y se lista aparte en la planilla y en la propuesta'; -- Control: opcionales por presupuesto SELECT presupuesto, count(*) FILTER (WHERE opcional) AS opcionales, count(*) FILTER (WHERE NOT opcional) AS del_presupuesto, sum(precio_total) FILTER (WHERE opcional) AS monto_opcional, sum(precio_total) FILTER (WHERE NOT opcional) AS monto_base FROM presupuesto_rubros GROUP BY 1 HAVING count(*) FILTER (WHERE opcional) > 0 ORDER BY 1;
